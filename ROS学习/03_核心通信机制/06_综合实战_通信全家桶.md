# 06 - 综合实战：通信全家桶工程

> 03 章毕业项目：一个工程把 Topic / Service / Action / 参数服务器 / 自定义消息全部串起来。
> 所有代码为 ROS2 Humble + rclcpp 风格，可直接作为真实项目的脚手架。

## 2️⃣ robot_node.cpp：被动提供方（一节点三件套 + 参数）

```cpp
// 机器人本体节点：发布状态(Topic) + 提供服务(Service) + 执行导航(Action) + 暴露参数
#include <rclcpp/rclcpp.hpp>
#include <rclcpp_action/rclcpp_action.hpp>
#include <robot_msgs/msg/robot_status.hpp>
#include <robot_msgs/srv/get_robot_info.hpp>
#include <robot_msgs/action/navigate_to.hpp>

using namespace std::placeholders;
using RobotStatus = robot_msgs::msg::RobotStatus;
using GetRobotInfo = robot_msgs::srv::GetRobotInfo;
using NavigateTo = robot_msgs::action::NavigateTo;
using GoalHandleNav = rclcpp_action::ServerGoalHandle<NavigateTo>;

class RobotNode : public rclcpp::Node {
public:
    RobotNode() : Node("robot_node") {
        // ===== ① 参数服务器：声明 + 动态调参回调 =====
        declare_parameter("max_speed", 1.0);   // m/s
        declare_parameter("kp", 0.5);          // 控制参数，演示动态调参
        param_cb_ = add_on_set_parameters_callback(
            std::bind(&RobotNode::onParamChange, this, _1));

        // ===== ② Topic 发布者：1Hz 播报状态 =====
        status_pub_ = create_publisher<RobotStatus>("robot_status", 10);
        status_timer_ = create_wall_timer(std::chrono::seconds(1),
            std::bind(&RobotNode::publishStatus, this));

        // ===== ③ Service 服务端：回答查询 =====
        info_srv_ = create_service<GetRobotInfo>("get_robot_info",
            std::bind(&RobotNode::handleInfoQuery, this, _1, _2));

        // ===== ④ Action 服务端：执行导航（三回调） =====
        nav_server_ = rclcpp_action::create_server<NavigateTo>(
            this, "navigate_to",
            std::bind(&RobotNode::handleGoal,     this, _1, _2),
            std::bind(&RobotNode::handleCancel,   this, _1),
            std::bind(&RobotNode::handleAccepted, this, _1));
    }

private:
    // ---------- ① 参数 ----------
    rclcpp::node_interfaces::OnSetParametersCallbackHandle::SharedPtr param_cb_;

    rcl_interfaces::msg::SetParametersResult onParamChange(
        const std::vector<rclcpp::Parameter>& params) {
        for (const auto& p : params)
            RCLCPP_INFO(get_logger(), "Param changed: %s", p.get_name().c_str());
        rcl_interfaces::msg::SetParametersResult result;
        result.successful = true;
        return result;
    }

    // ---------- ② Topic ----------
    rclcpp::Publisher<RobotStatus>::SharedPtr status_pub_;
    rclcpp::TimerBase::SharedPtr status_timer_;
    float battery_ = 100.0;

    void publishStatus() {
        RobotStatus msg;
        msg.robot_id = "robot_001";
        msg.battery_percent = battery_ = std::max(0.0f, battery_ - 0.1f);
        msg.is_moving = false;
        status_pub_->publish(msg);
    }

    // ---------- ③ Service ----------
    rclcpp::Service<GetRobotInfo>::SharedPtr info_srv_;

    void handleInfoQuery(
        const std::shared_ptr<GetRobotInfo::Request> req,
        std::shared_ptr<GetRobotInfo::Response> res) {
        RCLCPP_INFO(get_logger(), "Info query: %s", req->query.c_str());
        res->found = true;
        res->name = "robot_001";
        res->battery = battery_;
    }

    // ---------- ④ Action ----------
    rclcpp_action::Server<NavigateTo>::SharedPtr nav_server_;

    rclcpp_action::GoalResponse handleGoal(
        const rclcpp_action::GoalUUID&,
        std::shared_ptr<const NavigateTo::Goal> goal) {
        RCLCPP_INFO(get_logger(), "Nav goal: (%.1f, %.1f)", goal->x, goal->y);
        return rclcpp_action::GoalResponse::ACCEPT_AND_EXECUTE;
    }

    rclcpp_action::CancelResponse handleCancel(
        const std::shared_ptr<GoalHandleNav>) {
        return rclcpp_action::CancelResponse::ACCEPT;
    }

    void handleAccepted(const std::shared_ptr<GoalHandleNav> gh) {
        std::thread{std::bind(&RobotNode::executeNav, this, gh)}.detach();
    }

    void executeNav(const std::shared_ptr<GoalHandleNav> gh) {
        auto result = std::make_shared<NavigateTo::Result>();
        for (int i = 0; i <= 100 && rclcpp::ok(); ++i) {
            if (gh->is_canceling()) {              // 轮询取消
                result->success = false;
                result->message = "canceled";
                gh->canceled(result);
                return;
            }
            auto fb = std::make_shared<NavigateTo::Feedback>();   // 周期反馈
            fb->progress = i / 100.0;
            gh->publish_feedback(fb);
            std::this_thread::sleep_for(std::chrono::milliseconds(50));
        }
        result->success = true;
        result->message = "arrived";
        gh->succeed(result);
    }
};

int main(int argc, char** argv) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<RobotNode>());
    rclcpp::shutdown();
    return 0;
}
```

**一个节点 = 三件套 + 参数**，对比记忆：

| 机制 | 创建（板斧） | 占位符 | 角色 |
|---|---|---|---|
| Topic | `create_publisher` + timer | 无（主动） | 闹钟：定时发 |
| Service | `create_service` | `_1,_2`（请求,响应） | 柜员：收请求填响应 |
| Action | `create_server` 三回调 | `_1,_2` / `_1` / `_1` | 前台+车间：接单派活，后台干 |
| 参数 | `declare_parameter` + `add_on_set_parameters_callback` | `_1` | 公告栏 |

## 3️⃣ monitor_node.cpp：主动使用方

```cpp
// 监控终端：订阅状态 + 查服务 + 发导航目标（发完 2 秒取消）
#include <rclcpp/rclcpp.hpp>
#include <rclcpp_action/rclcpp_action.hpp>
#include <robot_msgs/msg/robot_status.hpp>
#include <robot_msgs/srv/get_robot_info.hpp>
#include <robot_msgs/action/navigate_to.hpp>

using namespace std::placeholders;
using RobotStatus = robot_msgs::msg::RobotStatus;
using GetRobotInfo = robot_msgs::srv::GetRobotInfo;
using NavigateTo = robot_msgs::action::NavigateTo;
using GoalHandleNav = rclcpp_action::ClientGoalHandle<NavigateTo>;
using ClientT = rclcpp_action::Client<NavigateTo>;

class MonitorNode : public rclcpp::Node {
public:
    MonitorNode() : Node("monitor_node") {
        // ===== ① Topic 订阅 =====
        status_sub_ = create_subscription<RobotStatus>(
            "robot_status", 10,
            std::bind(&MonitorNode::onStatus, this, _1));

        // ===== ② Service 客户端 =====
        info_client_ = create_client<GetRobotInfo>("get_robot_info");
        // 等服务上线后查一次
        while (!info_client_->wait_for_service(std::chrono::seconds(1)))
            RCLCPP_INFO(get_logger(), "waiting for info service...");

        auto req = std::make_shared<GetRobotInfo::Request>();
        req->query = "who are you";
        auto future = info_client_->async_send_request(req);
        rclcpp::spin_until_future_complete(shared_from_this(), future);
        auto res = future.get();
        RCLCPP_INFO(get_logger(), "Robot: %s, battery %.1f%%",
                    res->name.c_str(), res->battery);

        // ===== ③ Action 客户端：发目标 + 2 秒后取消 =====
        nav_client_ = rclcpp_action::create_client<NavigateTo>(this, "navigate_to");
        while (!nav_client_->wait_for_action_server(std::chrono::seconds(1)))
            RCLCPP_INFO(get_logger(), "waiting for nav server...");
        sendNavGoal();
    }

private:
    // ---------- ① 订阅 ----------
    rclcpp::Subscription<RobotStatus>::SharedPtr status_sub_;

    void onStatus(const RobotStatus::SharedPtr msg) {
        RCLCPP_INFO(get_logger(), "[status] battery=%.1f%% moving=%d",
                    msg->battery_percent, msg->is_moving);
    }

    // ---------- ② 服务客户端 ----------
    rclcpp::Client<GetRobotInfo>::SharedPtr info_client_;

    // ---------- ③ Action 客户端 ----------
    ClientT::SharedPtr nav_client_;
    GoalHandleNav::SharedPtr goal_handle_;
    rclcpp::TimerBase::SharedPtr cancel_timer_;

    void sendNavGoal() {
        NavigateTo::Goal goal;
        goal.x = 5.0;  goal.y = 3.0;

        ClientT::SendGoalOptions options;
        options.goal_response_callback =
            [this](const GoalHandleNav::SharedPtr& gh) {
                if (gh) { RCLCPP_INFO(get_logger(), "goal accepted");
                          goal_handle_ = gh; }
                else      RCLCPP_ERROR(get_logger(), "goal rejected");
            };
        options.feedback_callback =
            [this](GoalHandleNav::SharedPtr,
                   const std::shared_ptr<const NavigateTo::Feedback>& fb) {
                RCLCPP_INFO(get_logger(), "progress %.0f%%", fb->progress * 100);
            };
        options.result_callback =
            [this](const GoalHandleNav::WrappedResult& r) {
                switch (r.code) {
                    case rclcpp_action::ResultCode::SUCCEEDED:
                        RCLCPP_INFO(get_logger(), "arrived!"); break;
                    case rclcpp_action::ResultCode::CANCELED:
                        RCLCPP_INFO(get_logger(), "canceled"); break;
                    default:
                        RCLCPP_ERROR(get_logger(), "failed"); break;
                }
            };

        nav_client_->async_send_goal(goal, options);

        cancel_timer_ = create_wall_timer(std::chrono::seconds(2), [this]() {
            if (goal_handle_)
                nav_client_->async_cancel_goal(goal_handle_);
        });
    }
};

int main(int argc, char** argv) {
    rclcpp::init(argc, argv);
    rclcpp::spin(std::make_shared<MonitorNode>());
    rclcpp::shutdown();
    return 0;
}
```

## ❓ 这套工程回答了什么问题

**Q: 五个机制怎么选？**
A: 看数据流向和时长——高频流（状态）→ Topic；短查询（信息）→ Service；长任务带进度（导航）→ Action；配置（速度上限）→ 参数；数据结构 → 自定义消息。

**Q: 一个真实 ROS 节点长什么样？**
A: 就是 `robot_node.cpp` 的样子——一个类里同时挂着 pub/sub/service/action/param，main 永远只有四件套。

**Q: 代码骨架哪里找？**
A: [ROS速查笔记](../../ROS速查笔记.md) 的"三兄弟对比" + 本文件的两个完整节点，覆盖 03 章全部代码模式。
