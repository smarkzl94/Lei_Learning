### 1. 激光雷达数据处理

#### 障碍物检测
```cpp
#include <sensor_msgs/LaserScan.h>
#include <geometry_msgs/Twist.h>

class ObstacleDetector {
public:
    void scanCallback(const sensor_msgs::LaserScan::ConstPtr& scan) {
        float min_front_dist = scan->range_max;
        bool obstacle_detected = false;
        
        // 检查前方扇形区域（-30° 到 +30°）
        for (size_t i = 0; i < scan->ranges.size(); ++i) {
            float angle = scan->angle_min + i * scan->angle_increment;
            
            if (std::abs(angle) < M_PI / 6) {  // ±30度
                float range = scan->ranges[i];
                if (range > scan->range_min && range < min_front_dist) {
                    min_front_dist = range;
                }
                if (range < 0.5) {
                    obstacle_detected = true;
                }
            }
        }
        
        // 控制决策
        geometry_msgs::Twist cmd;
        if (obstacle_detected) {
            cmd.linear.x = 0.0;
            cmd.angular.z = 0.5;  // 旋转避障
        } else {
            cmd.linear.x = 0.2;
        }
        cmd_pub.publish(cmd);
    }
    
private:
    ros::Publisher cmd_pub;
};
```

#### 简单跟随
```cpp
// 跟随前方最近的物体
void followObject(const sensor_msgs::LaserScan::ConstPtr& scan) {
    float min_dist = scan->range_max;
    float min_angle = 0;
    
    for (size_t i = 0; i < scan->ranges.size(); ++i) {
        float range = scan->ranges[i];
        if (range > scan->range_min && range < scan->range_max && range < min_dist) {
            min_dist = range;
            min_angle = scan->angle_min + i * scan->angle_increment;
        }
    }
    
    geometry_msgs::Twist cmd;
    if (min_dist > 1.0) {
        cmd.linear.x = 0.2;  // 前进
    } else {
        cmd.linear.x = 0.0;  // 停止
    }
    cmd.angular.z = 0.5 * min_angle;  // 转向目标
    
    cmd_pub.publish(cmd);
}
```

### 2. 图像处理基础

#### 颜色检测
```cpp
#include <cv_bridge/cv_bridge.h>
#include <opencv2/imgproc/imgproc.hpp>

void colorDetection(const sensor_msgs::Image::ConstPtr& msg) {
    cv_bridge::CvImagePtr cv_ptr = cv_bridge::toCvCopy(msg, "bgr8");
    cv::Mat hsv;
    cv::cvtColor(cv_ptr->image, hsv, cv::COLOR_BGR2HSV);
    
    // 检测红色物体
    cv::Mat mask1, mask2, mask;
    cv::inRange(hsv, cv::Scalar(0, 100, 100), cv::Scalar(10, 255, 255), mask1);
    cv::inRange(hsv, cv::Scalar(160, 100, 100), cv::Scalar(180, 255, 255), mask2);
    cv::bitwise_or(mask1, mask2, mask);
    
    // 找轮廓
    std::vector<std::vector<cv::Point>> contours;
    cv::findContours(mask, contours, cv::RETR_EXTERNAL, cv::CHAIN_APPROX_SIMPLE);
    
    for (const auto& contour : contours) {
        if (cv::contourArea(contour) > 1000) {
            cv::Rect rect = cv::boundingRect(contour);
            cv::rectangle(cv_ptr->image, rect, cv::Scalar(0, 255, 0), 2);
        }
    }
    
    cv::imshow("Detection", cv_ptr->image);
    cv::waitKey(1);
}
```

    pass.setFilterFieldName("z");

## ❓ 常见问题

**Q: 激光雷达和摄像头怎么配准？**
A: 需要进行外参标定，找到两个传感器坐标系之间的变换关系（旋转 + 平移）。

**Q: 点云处理太卡？**
A: 1) 降采样（voxel filter）；2) 只处理 ROI；3) 降低处理频率。

**Q: 没有 PCL 经验怎么学？**
A: PCL 官方教程 + ROS 中的 pcl_conversions 包。从简单的滤波和分割开始。
