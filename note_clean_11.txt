## 📚 核心知识点

### 1. 为什么需要智能指针
```cpp
// 原始指针的问题：容易内存泄漏
void risky() {
    int* p = new int(10);
    if (some_condition) return;  // 忘记 delete！内存泄漏
    delete p;
}

// 智能指针自动管理生命周期
void safe() {
    std::unique_ptr<int> p(new int(10));
    if (some_condition) return;  // 自动 delete，不会泄漏
}
```

### 2. unique_ptr（独占所有权）
```cpp
#include <memory>

// 创建
std::unique_ptr<int> p1(new int(10));
auto p2 = std::make_unique<int>(20);  // C++14，推荐

// 访问
*p1 = 100;
std::cout << *p2;

// 转移所有权
std::unique_ptr<int> p3 = std::move(p1);  // p1 变为空
// p1 现在为 nullptr

// 释放
p3.reset();           // 显式释放
int* raw = p3.release();  // 释放所有权，返回原始指针（需手动 delete）

// 数组版本
auto arr = std::make_unique<int[]>(10);  // 管理数组
arr[0] = 1;
```

### 3. shared_ptr（共享所有权）
```cpp
// 创建
std::shared_ptr<int> p1(new int(10));
auto p2 = std::make_shared<int>(20);  // 推荐，更高效

// 共享所有权
std::shared_ptr<int> p3 = p1;  // 引用计数 +1
std::cout << p1.use_count();   // 输出 2

// 引用计数为 0 时自动释放
{
    auto p4 = p3;  // 引用计数 +1
}  // p4 销毁，引用计数 -1

// 检查是否唯一
if (p1.unique()) {
    // 当前只有一个 owner
}
```

### 4. weak_ptr（弱引用）
```cpp
// 解决循环引用问题
struct B;
struct A {
    std::shared_ptr<B> b_ptr;
};
struct B {
    std::shared_ptr<A> a_ptr;  // 循环引用！两个对象都释放不了
};

// 正确做法：一方用 weak_ptr
struct B {
    std::weak_ptr<A> a_ptr;
};

// 使用 weak_ptr
std::shared_ptr<int> sp = std::make_shared<int>(10);
std::weak_ptr<int> wp = sp;

if (auto locked = wp.lock()) {  // 尝试提升为 shared_ptr
    // 对象还存在
    std::cout << *locked;
} else {
    // 对象已被释放
}

// 检查是否过期
if (wp.expired()) {
    // 对象已不存在
}
```

### 5. 自定义删除器
```cpp
// 文件关闭删除器
auto file_deleter = [](FILE* f) { fclose(f); };
std::unique_ptr<FILE, decltype(file_deleter)> 
    file(fopen("test.txt", "r"), file_deleter);

// 数组删除（C++17 前）
struct ArrayDeleter {
    void operator()(int* p) { delete[] p; }
};
std::unique_ptr<int, ArrayDeleter> arr(new int[10]);
```

### 6. 智能指针与函数
```cpp
// 参数传递
void process(std::shared_ptr<int> p);        // 共享所有权
void observe(const std::shared_ptr<int>& p); // 只观察，不增加引用计数
void take(std::unique_ptr<int> p);           // 转移所有权

// 返回值
std::unique_ptr<int> create();  // 工厂函数
```

## 💻 代码示例

```cpp
#include <iostream>
#include <memory>
#include <vector>

class Resource {
public:
    Resource(int id) : id(id) { 
        std::cout << "Resource " << id << " created" << std::endl; 
    }
    ~Resource() { 
        std::cout << "Resource " << id << " destroyed" << std::endl; 
    }
    void use() { std::cout << "Using resource " << id << std::endl; }
private:
    int id;
};

int main() {
    // unique_ptr 示例
    {
        auto r1 = std::make_unique<Resource>(1);
        r1->use();
    }  // 自动销毁
    
    // shared_ptr 示例
    std::shared_ptr<Resource> r2;
    {
        auto r3 = std::make_shared<Resource>(2);
        r2 = r3;  // 共享所有权
        std::cout << "Count: " << r3.use_count() << std::endl;  // 2
    }  // r3 销毁，但 r2 还在，Resource 2 不销毁
    std::cout << "Count: " << r2.use_count() << std::endl;  // 1
    
    // 容器中使用智能指针
    std::vector<std::shared_ptr<Resource>> resources;
    resources.push_back(std::make_shared<Resource>(3));
    resources.push_back(std::make_shared<Resource>(4));
    
    return 0;
}
```
