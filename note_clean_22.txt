## 📚 核心知识点

### 1. 函数模板
```cpp
// 通用加法函数
template <typename T>
T add(T a, T b) {
    return a + b;
}

// 使用
int sum1 = add(1, 2);           // T 推导为 int
double sum2 = add(1.5, 2.5);    // T 推导为 double
auto sum3 = add<std::string>("Hello, ", "World!");  // 显式指定

// 多模板参数
template <typename T, typename U>
auto multiply(T a, U b) -> decltype(a * b) {
    return a * b;
}
```

### 2. 类模板
```cpp
template <typename T>
class Stack {
public:
    void push(const T& value) {
        data.push_back(value);
    }
    
    void pop() {
        if (!data.empty()) data.pop_back();
    }
    
    T& top() {
        return data.back();
    }
    
    bool empty() const {
        return data.empty();
    }
    
private:
    std::vector<T> data;
};

// 使用
Stack<int> int_stack;
Stack<std::string> string_stack;
```

### 3. 模板特化
```cpp
// 通用版本
template <typename T>
class Comparator {
public:
    bool equal(T a, T b) {
        return a == b;
    }
};

// 全特化：针对 const char*
template <>
class Comparator<const char*> {
public:
    bool equal(const char* a, const char* b) {
        return strcmp(a, b) == 0;
    }
};

// 偏特化：针对指针类型
template <typename T>
class Comparator<T*> {
public:
    bool equal(T* a, T* b) {
        return *a == *b;  // 比较指向的内容
    }
};
```

### 4. 变参模板（C++11）
```cpp
// 递归终止
template <typename T>
T sum(T value) {
    return value;
}

// 变参展开
template <typename T, typename... Args>
T sum(T first, Args... args) {
    return first + sum(args...);
}

// 使用
int result = sum(1, 2, 3, 4, 5);  // 15

// C++17 折叠表达式
template <typename... Args>
auto sum_fold(Args... args) {
    return (args + ...);  // 一元右折叠
}
```

### 5. 类型特征（C++11）
```cpp
#include <type_traits>

// 编译期类型判断
std::is_integral<int>::value;        // true
std::is_pointer<int*>::value;        // true
std::is_reference<int&>::value;      // true

// enable_if：条件模板启用
template <typename T>
typename std::enable_if<std::is_integral<T>::value, T>::type
foo(T value) {
    return value * 2;
}

// C++14 简化
std::is_integral_v<int>;  // 等价于 std::is_integral<int>::value
```

### 6. 概念（Concepts，C++20）
```cpp
template <typename T>
concept Addable = requires(T a, T b) {
    { a + b } -> std::convertible_to<T>;
};

template <Addable T>
T add(T a, T b) {
    return a + b;
}
```

## 💻 代码示例

```cpp
#include <iostream>
#include <vector>
#include <type_traits>

// 泛型打印函数
template <typename Container>
void print(const Container& c) {
    for (const auto& elem : c) {
        std::cout << elem << " ";
    }
    std::cout << std::endl;
}

// 编译期 if（C++17）
template <typename T>
auto get_value(T t) {
    if constexpr (std::is_pointer_v<T>) {
        return *t;  // 如果 T 是指针，解引用
    } else {
        return t;   // 否则直接返回
    }
}

// 类型安全的数组大小
template <typename T, size_t N>
size_t array_size(const T (&)[N]) {
    return N;
}

int main() {
    std::vector<int> vec = {1, 2, 3, 4, 5};
    print(vec);
    
    int x = 10;
    int* p = &x;
    std::cout << get_value(x) << std::endl;  // 10
    std::cout << get_value(p) << std::endl;  // 10
    
    int arr[] = {1, 2, 3, 4, 5};
    std::cout << array_size(arr) << std::endl;  // 5
    
    return 0;
}
```
