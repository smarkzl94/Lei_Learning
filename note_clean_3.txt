## 📚 核心知识点

### 1. 函数基础

#### 定义与声明
```cpp
// 函数声明（原型）
int add(int a, int b);

// 函数定义
int add(int a, int b) {
    return a + b;
}

// 调用
int result = add(3, 5);  // result = 8
```

#### 返回类型推导（C++14）
```cpp
auto multiply(int a, int b) {  // 返回类型自动推导为 int
    return a * b;
}
```

### 2. 参数传递方式

```cpp
// 1. 值传递：实参的副本传入，函数内修改不影响原值
void swap_by_value(int a, int b) {
    int temp = a;
    a = b;
    b = temp;  // 交换的是副本，原变量不变
}

// 2. 引用传递：直接操作原变量
void swap_by_ref(int& a, int& b) {
    int temp = a;
    a = b;
    b = temp;  // 原变量被交换
}

// 3. 指针传递
void swap_by_ptr(int* a, int* b) {
    int temp = *a;
    *a = *b;
    *b = temp;
}

// 使用
int x = 3, y = 5;
swap_by_value(x, y);   // x=3, y=5（不变）
swap_by_ref(x, y);     // x=5, y=3（交换）
swap_by_ptr(&x, &y);   // x=3, y=5（再次交换）
```

### 3. 默认参数
```cpp
// 默认参数必须从右向左连续
void print(int x, int y = 10, int z = 20);

print(1);        // 等价于 print(1, 10, 20)
print(1, 2);     // 等价于 print(1, 2, 20)
print(1, 2, 3);  // 等价于 print(1, 2, 3)
```

### 4. 函数重载
```cpp
// 同名函数，参数列表不同
int add(int a, int b);
double add(double a, double b);
int add(int a, int b, int c);

// 编译器根据参数类型/数量自动选择
add(1, 2);        // 调用 int 版本
add(1.5, 2.5);    // 调用 double 版本
add(1, 2, 3);     // 调用三参数版本
```

### 5. 内联函数
```cpp
// 内联：建议编译器将函数体直接嵌入调用处（减少函数调用开销）
inline int square(int x) {
    return x * x;
}

// 现代 C++ 中，编译器会自动决定是否内联，inline 只是建议
```

### 6. 递归函数
```cpp
// 计算阶乘
int factorial(int n) {
    if (n <= 1) return 1;           // 基准情况
    return n * factorial(n - 1);    // 递归调用
}

// 斐波那契数列
int fibonacci(int n) {
    if (n <= 1) return n;
    return fibonacci(n - 1) + fibonacci(n - 2);
}
```

### 7. Lambda 表达式（C++11）
```cpp
// 基本语法：[捕获列表](参数列表) -> 返回类型 { 函数体 }
auto add = [](int a, int b) -> int {
    return a + b;
};

// 捕获外部变量
int x = 10;
auto capture_by_value = [x]() { return x; };      // 拷贝 x
auto capture_by_ref = [&x]() { return ++x; };     // 引用 x
auto capture_all_by_ref = [&]() { /* ... */ };    // 引用捕获所有
auto capture_all_by_value = [=]() { /* ... */ };  // 拷贝捕获所有
```

## 💻 代码示例

```cpp
#include <iostream>
#include <vector>
#include <algorithm>

// 函数重载示例
int max(int a, int b) {
    return (a > b) ? a : b;
}

double max(double a, double b) {
    return (a > b) ? a : b;
}

// 使用引用来返回多个值
void get_min_max(const std::vector<int>& arr, int& min_val, int& max_val) {
    if (arr.empty()) return;
    min_val = max_val = arr[0];
    for (size_t i = 1; i < arr.size(); ++i) {
        if (arr[i] < min_val) min_val = arr[i];
        if (arr[i] > max_val) max_val = arr[i];
    }
}

int main() {
    // 函数重载
    std::cout << max(3, 5) << std::endl;
    std::cout << max(3.14, 2.71) << std::endl;
    
    // Lambda 排序
    std::vector<int> nums = {3, 1, 4, 1, 5, 9};
    std::sort(nums.begin(), nums.end(), [](int a, int b) {
        return a > b;  // 降序
    });
    
    // 引用返回多个值
    int min_v, max_v;
    get_min_max(nums, min_v, max_v);
    std::cout << "Min: " << min_v << ", Max: " << max_v << std::endl;
    
    return 0;
}
```
