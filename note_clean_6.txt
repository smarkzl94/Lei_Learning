## 📚 核心知识点

### 1. 类的定义
```cpp
class Student {
public:              // 公有成员：外部可访问
    // 成员函数（方法）
    void setName(const std::string& n);
    std::string getName() const;
    void printInfo() const;

private:             // 私有成员：仅类内部可访问
    // 成员变量（属性）
    std::string name;
    int age;
    double score;
};

// 成员函数实现
void Student::setName(const std::string& n) {
    name = n;
}

std::string Student::getName() const {
    return name;
}

void Student::printInfo() const {
    std::cout << "Name: " << name << ", Age: " << age << std::endl;
}
```

### 2. 对象的创建与使用
```cpp
// 栈上创建对象
Student stu1;
stu1.setName("Alice");   // 使用 . 访问成员

// 堆上创建对象
Student* stu2 = new Student();
stu2->setName("Bob");    // 使用 -> 访问成员
delete stu2;             // 必须手动释放

// C++11 统一初始化
Student stu3{"Charlie", 20, 85.5};
```

### 3. 访问控制

| 访问修饰符 | 类内部 | 子类 | 外部 |
|-----------|--------|------|------|
| `public` | ✓ | ✓ | ✓ |
| `protected` | ✓ | ✓ | ✗ |
| `private` | ✓ | ✗ | ✗ |

### 4. struct vs class
```cpp
// struct 默认 public
struct Point {
    double x;  // 默认 public
    double y;
};

// class 默认 private
class Point2 {
    double x;  // 默认 private
    double y;
};
```
> 本质相同，只是默认访问权限不同。通常用 `struct` 表示纯数据结构，用 `class` 表示有封装行为的抽象。

### 5. this 指针
```cpp
class Person {
public:
    void setName(const std::string& name) {
        this->name = name;  // this->name 是成员变量，name 是参数
    }
    
    Person* getThis() {
        return this;  // 返回当前对象的指针
    }
    
private:
    std::string name;
};
```

### 6. 静态成员
```cpp
class Counter {
public:
    Counter() {
        ++count;  // 每创建一个对象，计数增加
    }
    
    static int getCount() {  // 静态成员函数
        return count;
    }
    
private:
    static int count;  // 静态成员变量声明
};

int Counter::count = 0;  // 静态成员变量定义和初始化

// 使用
Counter c1, c2;
std::cout << Counter::getCount();  // 输出 2
```

### 7. 常量成员函数
```cpp
class Date {
public:
    int getYear() const {   // const 成员函数：不修改对象状态
        return year;
    }
    
    void setYear(int y) {   // 非 const：可以修改
        year = y;
    }
    
private:
    int year;
};

const Date d{2024};
d.getYear();   // OK：const 对象只能调用 const 成员函数
// d.setYear(2025);  // 错误！
```

## 💻 代码示例

```cpp
#include <iostream>
#include <string>

class Rectangle {
public:
    // 构造函数
    Rectangle(double w, double h) : width(w), height(h) {}
    
    // 默认参数构造函数
    Rectangle() : Rectangle(1.0, 1.0) {}  // 委托构造函数（C++11）
    
    // 成员函数
    double getArea() const {
        return width * height;
    }
    
    double getPerimeter() const {
        return 2 * (width + height);
    }
    
    void setWidth(double w) {
        if (w > 0) width = w;
    }
    
    void setHeight(double h) {
        if (h > 0) height = h;
    }
    
private:
    double width;
    double height;
};

int main() {
    Rectangle rect{5.0, 3.0};
    std::cout << "Area: " << rect.getArea() << std::endl;
    std::cout << "Perimeter: " << rect.getPerimeter() << std::endl;
    
    return 0;
}
```
